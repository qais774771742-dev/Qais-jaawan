import 'person.dart';
class Student extends Person {
  late int level;
  late double degree;
  Student({
    required super.id,
    required super.name,
    required super.department,
    required super.age,
    required this.level,
    required this.degree,
  });
  @override
  void printDetails() {
    print("${name}      ${age}     ${level}     ${department}     ${degree}");
  }
}
